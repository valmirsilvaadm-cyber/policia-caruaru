rules_version = '2';

// ══════════════════════════════════════════════════════════════════════════
// REGRAS DE SEGURANÇA — Polícia Municipal de Caruaru (app)
// Versão 2 — agora com Firebase Authentication para TODOS os usuários
// (não só o administrador).
//
// O que mudou em relação à versão anterior:
//  ✅ Todo mundo que usa o app agora tem uma sessão real do Firebase
//     (request.auth != null deixa de ser sempre falso para o usuário comum).
//  ✅ Por isso, praticamente toda a leitura/escrita passou a exigir
//     "request.auth != null" — alguém sem conta nenhuma no app (nem
//     aprovada, nem pendente) não consegue mais ler ou escrever nada
//     direto no banco, mesmo com acesso técnico ao Firebase SDK.
//  ✅ O cadastro de um documento em "usuarios" só pode ser feito pela
//     própria pessoa dona daquele UID (request.auth.uid == userId),
//     nunca em nome de outra conta.
//
// O que ainda NÃO é garantido (limitação aceitável, pois exigiria back-end):
//  ⚠️ Qualquer conta aprovada consegue ler dados de módulos aos quais não
//     tem permissão na INTERFACE (ex: um usuário só de Inspetoria consegue,
//     via acesso direto ao banco, ler documentos de "pad"/Corregedoria).
//     A tela do app já esconde isso, mas o banco, por si, não impede um
//     usuário autenticado (ainda que aprovado) de ler outras coleções.
//     Para bloquear isso por módulo seria necessário guardar as permissões
//     em "custom claims" do Firebase Auth (isso exige Cloud Functions,
//     que não fazem parte deste projeto hoje) e reescrever cada regra para
//     checar a permissão específica.
// ══════════════════════════════════════════════════════════════════════════

service cloud.firestore {
  match /databases/{database}/documents {

    // ─── Helpers ───
    function logado() {
      return request.auth != null;
    }
    function isAdmin() {
      return logado() &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // ─── Coleção interna que define quem é admin. Gerenciada só pelo console. ───
    match /admins/{uid} {
      allow read, write: if false;
    }

    // ─── USUÁRIOS (cadastro, aprovação, permissões) ───
    match /usuarios/{userId} {
      // precisa estar logado (mesmo pendente/recém-cadastrado) para ler —
      // usado pelo próprio login e por telas que listam usuários por permissão.
      allow read: if logado();

      // só pode criar o PRÓPRIO documento (mesmo UID da conta recém-criada),
      // e nunca já nascendo aprovado ou com permissão sensível.
      allow create: if logado() && request.auth.uid == userId &&
        request.resource.data.status == "pendente" &&
        request.resource.data.permissoes.comando == false &&
        request.resource.data.permissoes.corregedoria == false &&
        request.resource.data.permissoes.administrativo == false &&
        request.resource.data.permissoes.patrimonio == false;

      // aprovar, mudar permissões e editar — só admin.
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // ─── SERVIDORES (roster usado por plantão/O.S/ficha) ───
    match /servidores_plantao/{id} {
      allow read: if logado();
      allow create, update: if logado();
      allow delete: if isAdmin();
    }

    // ─── AUDITORIA — trilha somente-inserção ───
    match /auditoria/{id} {
      allow read: if logado();
      allow create: if logado();
      allow update: if false;
      allow delete: if isAdmin();
    }

    // ─── LOG DE ACESSOS — trilha somente-inserção ───
    // create fica liberado mesmo sem login, pois é usado para registrar
    // TENTATIVAS de login que falharam (aí ainda não existe request.auth).
    match /log_acessos/{id} {
      allow read: if isAdmin();
      allow create: if true;
      allow update: if false;
      allow delete: if isAdmin();
    }

    // ─── PLANTÃO ───
    match /plantao/{id} {
      allow read: if logado();
      allow create, update: if logado();
      allow delete: if isAdmin();
    }

    // ─── FISCALIZAÇÃO ───
    match /fiscalizacao/{id} {
      allow read: if logado();
      allow create, update: if logado();
      allow delete: if isAdmin();
    }

    // ─── ORDENS DE SERVIÇO ───
    match /ordens_servico/{id} {
      allow read: if logado();
      allow create, update: if logado();
      allow delete: if isAdmin();
    }

    // ─── OCORRÊNCIAS ───
    match /ocorrencias/{id} {
      allow read: if logado();
      allow create, update: if logado();
      allow delete: if isAdmin();
    }

    match /ocorrencia_historico/{id} {
      allow read: if logado();
      allow create: if logado();
      allow update: if false;
      allow delete: if isAdmin();
    }

    // ─── PAD (Corregedoria) ───
    match /pad/{id} {
      allow read: if logado();
      allow create, update: if logado();
      allow delete: if isAdmin();
    }

    match /pad_historico/{id} {
      allow read: if logado();
      allow create: if logado();
      allow update: if false;
      allow delete: if isAdmin();
    }

    // ─── LINHA DO TEMPO DO SERVIDOR ───
    match /servidor_timeline/{id} {
      allow read: if logado();
      allow create: if logado();
      allow update: if false;
      allow delete: if isAdmin();
    }

    // ─── NOTIFICAÇÕES (sino) ───
    match /notificacoes/{id} {
      allow read: if logado();
      allow create: if logado();
      // só permite marcar como lida — nunca trocar destinatário/conteúdo
      allow update: if logado() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(["lida"]);
      allow delete: if logado();
    }

    // ─── Qualquer coleção não listada acima (ex: config_admin) fica
    // acessível apenas pelo Administrador — que tem acesso a tudo no app.
    match /{document=**} {
      allow read, write: if isAdmin();
    }
  }
}
