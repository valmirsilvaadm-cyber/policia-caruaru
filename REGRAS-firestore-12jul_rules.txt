rules_version = '2';

// ======================================================================
// REGRAS DE SEGURANCA - SISTEMA POLICIA MUNICIPAL DE CARUARU
// ======================================================================
// Como publicar:
// 1. Console do Firebase -> Firestore Database -> Regras
// 2. Apague todo o conteudo do editor
// 3. Cole todo o conteudo deste arquivo
// 4. Clique em "Publicar"
//
// Modelo de acesso:
// - E' preciso estar autenticado (Firebase Authentication) para ler ou
//   escrever qualquer coisa. Nao ha leitura publica.
// - "usuarios/{uid}" guarda o cadastro de cada servidor, com:
//     status:      "pendente" | "aprovado" | "rejeitado"
//     permissoes:  mapa por modulo (checklist, inspetoria, administrativo,
//                  videomonitor, inteligencia, comando, patrimonio,
//                  tecnologia, fotos, escala, dashboard, corregedoria).
//                  O padrao e liberado; so bloqueia se o campo estiver
//                  marcado como "false" (mesmo comportamento do app).
// - "admins/{uid}" marca quem e administrador geral. So e criado ou
//   alterado manualmente pelo console - o app somente le esse documento.
// - Registros de auditoria e historico sao "somente cria" (ninguem edita
//   ou apaga pelo app, garantindo rastreabilidade).
// ======================================================================

service cloud.firestore {
  match /databases/{database}/documents {

    // -- FUNCOES AUXILIARES --------------------------------------------
    function logado() {
      return request.auth != null;
    }

    function meuUsuario() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data;
    }

    // Tecnologia agora acumula tudo que era exclusivo do super-admin secreto —
    // não existe mais uma conta de administrador separada no app. A coleção
    // "admins" continua funcionando como reforço/plano B (acesso direto via
    // Console, sem depender do app), mas deixou de ser o único caminho.
    function souAdmin() {
      return logado() &&
        (exists(/databases/$(database)/documents/admins/$(request.auth.uid)) ||
         temPermissao("tecnologia"));
    }

    function souAprovado() {
      return logado() &&
        exists(/databases/$(database)/documents/usuarios/$(request.auth.uid)) &&
        meuUsuario().status == "aprovado";
    }

    // Livre por padrao: so bloqueia se o modulo estiver explicitamente
    // marcado como "false" nas permissoes do usuario (espelha o app).
    function temPermissao(modulo) {
      return souAprovado() &&
        (!(modulo in meuUsuario().permissoes) ||
          meuUsuario().permissoes[modulo] != false);
    }

    function souDono(uid) {
      return logado() && request.auth.uid == uid;
    }

    // -- ADMINS (gestao so pelo console) --------------------------------
    match /admins/{uid} {
      allow read: if souDono(uid) || souAdmin();
      allow write: if false; // gerencie pelo console do Firebase
    }

    // -- USUARIOS (cadastro / aprovacao / permissoes) -------------------
    match /usuarios/{uid} {
      allow read: if souDono(uid) || souAdmin() || temPermissao("tecnologia");

      // Cadastro: a propria pessoa cria seu documento, sempre como
      // "pendente" e sem poder se autoaprovar ou se autoconceder
      // permissoes maiores que o padrao inicial.
      allow create: if souDono(uid) &&
        request.resource.data.status == "pendente";

      // A propria pessoa pode atualizar dados cadastrais, mas nunca
      // o proprio status ou as proprias permissoes. Tecnologia gerencia
      // contas de outros usuarios (aprovar, desativar, permissoes, etc).
      allow update: if souAdmin() || temPermissao("tecnologia") ||
        (souDono(uid) &&
          request.resource.data.status == resource.data.status &&
          request.resource.data.permissoes == resource.data.permissoes);

      allow delete: if souAdmin() || temPermissao("tecnologia");
    }

    // -- AUDITORIA E LOGS (somente cria - trilha integra) ---------------
    match /auditoria/{id} {
      allow read: if souAdmin() || temPermissao("comando") || temPermissao("corregedoria") || temPermissao("secretario");
      allow create: if souAprovado();
      allow update, delete: if false;
    }

    match /log_acessos/{id} {
      allow read: if souAdmin() || temPermissao("tecnologia");
      allow create: if logado();
      allow update, delete: if false;
    }

    // -- NOTIFICACOES (cada um so ve as suas) ---------------------------
    match /notificacoes/{id} {
      allow read: if souAdmin() ||
        (logado() && resource.data.destEmail == request.auth.token.email);
      allow create: if souAprovado();
      allow update: if souAdmin() ||
        (logado() && resource.data.destEmail == request.auth.token.email);
      allow delete: if souAdmin();
    }

    // -- ASSUNCAO DE PLANTAO ---------------------------------------------
    match /plantao/{id} {
      allow read: if souAprovado();
      allow create: if temPermissao("inspetoria");
      allow update: if souAdmin() || temPermissao("inspetoria") ||
        ( // Confirmação de assunção pelo próprio servidor — só uma vez, só após
          // autorização do inspetor, e só alterando os campos da confirmação.
          souAprovado() &&
          resource.data.servidorMat == meuUsuario().matricula &&
          resource.data.autorizadoAssuncao == true &&
          resource.data.confirmadoPeloServidor != true &&
          request.resource.data.confirmadoPeloServidor == true &&
          request.resource.data.confirmadoUid == request.auth.uid &&
          request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(["confirmadoPeloServidor","confirmadoUid","confirmadoNome","confirmadoEm","statusAssuncao"])
        );
      allow delete: if souAdmin();
    }

    match /plantao_fechamentos/{id} {
      allow read: if souAprovado();
      allow create: if temPermissao("inspetoria");
      allow update, delete: if souAdmin();
    }

    match /servidores_plantao/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("administrativo");
    }

    // -- OCORRENCIAS (Inspetoria -> Comando -> Corregedoria) -------------
    match /ocorrencias/{id} {
      allow read: if souAprovado();
      allow create: if souAprovado();
      // So quem decide (Comando/Corregedoria) ou administra pode alterar
      // o status/encaminhamento; quem registrou tambem pode atualizar.
      allow update: if souAdmin() || temPermissao("comando") || temPermissao("corregedoria") ||
        (souAprovado() && resource.data.email == request.auth.token.email);
      allow delete: if souAdmin();
    }

    match /ocorrencia_historico/{id} {
      allow read: if souAdmin() || temPermissao("comando") || temPermissao("corregedoria");
      allow create: if souAprovado();
      allow update, delete: if false;
    }

    // -- ORDENS DE SERVICO -------------------------------------------------
    match /ordens_servico/{id} {
      allow read: if souAprovado();
      allow create: if temPermissao("inspetoria");
      allow update: if souAdmin() || temPermissao("comando") || temPermissao("inspetoria");
      allow delete: if souAdmin();
    }

    // -- FISCALIZACAO --------------------------------------------------------
    match /fiscalizacao/{id} {
      allow read: if souAprovado();
      allow create: if temPermissao("inspetoria");
      allow update: if souAdmin() || temPermissao("inspetoria");
      allow delete: if souAdmin();
    }

    // -- CORREGEDORIA / PAD -------------------------------------------------
    match /pad/{id} {
      allow read: if souAdmin() || temPermissao("corregedoria") || temPermissao("secretario") ||
        (souAprovado() && resource.data.servidor == meuUsuario().nome);
      allow create: if temPermissao("corregedoria") || temPermissao("comando");
      allow update: if souAdmin() || temPermissao("corregedoria");
      allow delete: if souAdmin();
    }

    match /pad_historico/{id} {
      allow read: if souAdmin() || temPermissao("corregedoria");
      allow create: if souAprovado();
      allow update, delete: if false;
    }

    // -- ARMARIA -------------------------------------------------------------
    match /armas_individuais/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("checklist");
    }
    match /armas_historico/{id} {
      allow read: if souAprovado();
      allow create: if souAprovado();
      allow update, delete: if false;
    }
    match /coletes/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("checklist");
    }
    match /coletes_historico/{id} {
      allow read: if souAprovado();
      allow create: if souAprovado();
      allow update, delete: if false;
    }
    match /municoes/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("checklist");
    }
    match /municoes_historico/{id} {
      allow read: if souAprovado();
      allow create: if souAprovado();
      allow update, delete: if false;
    }

    // -- ADMINISTRATIVO (fotos, escala, banco de horas, timeline) -----------
    match /fotos/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("fotos");
    }
    match /escalas_mensais/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("escala") || temPermissao("administrativo");
    }
    match /equipes/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("inspetoria");
    }
    match /permutas/{id} {
      allow read: if souAprovado();
      allow create: if souAprovado();
      allow update: if souAdmin() || temPermissao("administrativo") ||
        ( // o próprio destinatário aceita ou recusa — só enquanto aguarda a resposta dele,
          // e só pode mover para esses dois status (não pode se auto-aprovar)
          souAprovado() &&
          resource.data.destinatarioMat == meuUsuario().matricula &&
          resource.data.status == "aguardando_servidor" &&
          request.resource.data.status in ["aguardando_administrativo","rejeitada_servidor"]
        );
      allow delete: if souAdmin();
    }
    match /banco_horas/{id} {
      allow read: if souAprovado();
      allow create: if souAdmin() || temPermissao("administrativo") || temPermissao("escala") || temPermissao("inspetoria");
      allow update: if souAdmin() || temPermissao("administrativo") || temPermissao("escala") ||
        ( // o inspetor só pode reenviar o PRÓPRIO lançamento que voltou pedindo correção —
          // não pode aprovar, rejeitar ou alterar lançamentos de outros
          temPermissao("inspetoria") &&
          resource.data.lancadoPorMat == meuUsuario().matricula &&
          resource.data.status == "correcao_solicitada" &&
          request.resource.data.status == "pendente_analise"
        );
      allow delete: if souAdmin();
    }
    match /servidor_timeline/{id} {
      allow read: if souAprovado();
      allow create: if souAdmin() || temPermissao("administrativo");
      allow update, delete: if souAdmin();
    }

    // -- PATRIMONIO / VIATURAS -----------------------------------------------
    match /viaturas/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("patrimonio") || temPermissao("inspetoria");
    }
    match /patrimonio/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("patrimonio");
    }
    match /patrimonio_historico/{id} {
      allow read: if souAprovado();
      allow create: if souAdmin() || temPermissao("patrimonio");
      allow update, delete: if false;
    }
    match /patrimonio_inventarios/{id} {
      allow read: if souAprovado();
      allow create: if souAdmin() || temPermissao("patrimonio");
      allow update, delete: if souAdmin();
    }

    // -- TECNOLOGIA -----------------------------------------------------------
    match /chamados_tecnicos/{id} {
      allow read: if souAprovado();
      allow create: if souAprovado();
      allow update: if souAdmin() || temPermissao("tecnologia");
      allow delete: if souAdmin();
    }
    match /sistema_config/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("tecnologia");
    }
    match /perfis_permissao/{id} {
      allow read: if souAprovado();
      allow write: if souAdmin() || temPermissao("tecnologia");
    }
    match /erros_sistema/{id} {
      allow read: if souAdmin() || temPermissao("tecnologia");
      allow create: if souAprovado();
      allow update: if souAdmin() || temPermissao("tecnologia");
      allow delete: if souAdmin();
    }
    match /dispositivos/{id} {
      allow read: if souAdmin() || temPermissao("tecnologia");
      allow write: if souAdmin() || temPermissao("tecnologia");
    }

    // -- FALLBACK - nega tudo que nao foi listado acima ----------------------
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
